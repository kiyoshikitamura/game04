'use client';
import {useMemo,useState} from 'react';
import BattleView from '@/app/components/redesign/BattleView';
import {AudioProvider} from '@/audio/AudioProvider';
import {createSkillVfxFixture} from '../skill-vfx24/fixture';
import chars from '@/theme/local-characters.json';
export default function Page(){
 const [id,setId]=useState('char_miyabi_01'),[count,setCount]=useState(3);
 const result=useMemo(()=>{const r=createSkillVfxFixture({skillId:'SKD007',side:'ally',burst:false,success:true,enemies:count,cutin:'SSR'}),c=chars.find(c=>c.id===id)!,image=c.battle||c.full;r.waves.forEach(w=>w.forEach(u=>{u.name=c.name;u.image=image}));r.frames.forEach(f=>f.enemies.forEach(u=>u.image=image));return r},[id,count]);
 return <AudioProvider><main style={{height:'100dvh',overflow:'auto'}}><select aria-label="character" value={id} onChange={e=>setId(e.target.value)}>{chars.map(c=><option key={c.id} value={c.id}>{c.name}</option>)}</select><select aria-label="count" value={count} onChange={e=>setCount(+e.target.value)}>{[1,2,3].map(n=><option key={n}>{n}</option>)}</select><BattleView key={id+count} result={result} initialFrame={1} initialPaused hideWaveDisplay vipActive onRetire={()=>{}} onComplete={()=>{}} /></main></AudioProvider>
}
